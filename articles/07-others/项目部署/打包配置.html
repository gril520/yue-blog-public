打包示例分析 ​

js
const path = require('path')
const BundleAnalyzerPlugin = require('webpack-bundle-analyzer').BundleAnalyzerPlugin
const TrRouterPlugin = require('@admin/tr-plugin-router')
const { projectName } = require('./src/settings')
const resolve = dir => path.join(__dirname, dir)
// page title
// 生产环境，测试和正式
const IS_PROD = ['production'].includes(process.env.NODE_ENV)
// externals
const externals = {
  vue: 'Vue',
  'vue-router': 'VueRouter',
  vuex: 'Vuex',
  vant: 'vant',
  axios: 'axios'
}

module.exports = {
  publicPath: '/baseplatformh5',
  outputDir: 'dist',
  assetsDir: 'static',
  filenameHashing: true,
  pages: {
    index: {
      entry: 'src/main.js',
      template: IS_PROD ? 'public/deploy.html' : 'public/dev.html',
      filename: 'index.html',
      title: projectName,
      chunks: 'all'
    }
  },
  devServer: {
    port: 9020, // 端口
    headers: {
      'Access-Control-Allow-Origin': '*'
    },
    proxy: {
      '/gateway': {
        target: 'http://newevntestbaseplatform.admin.com',
        // target: 'http://devbaseplatform.admin.com',
        changeOrigin: true,
        logLevel: 'debug',
        pathRewrite: {
          '^/gateway': '/gateway'
        }
      }
    },
    hot: true,
    client: {
      logging: 'error',
      overlay: false
    }
  },
  css: {
    extract: IS_PROD, // 是否将组件中的 CSS 提取至一个独立的 CSS 文件中 (而不是动态注入到 JavaScript 中的 inline 代码)。
    sourceMap: false,
    loaderOptions: {
      scss: {
        // 向全局sass样式传入共享的全局变量, $src可以配置图片cdn前缀
        // 详情: https://cli.vuejs.org/guide/css.html#passing-options-to-pre-processor-loaders
        prependData: `
          @import "@/assets/css/mixin.scss";
          `
      }
    }
  },
  configureWebpack: config => {
    config.externals = externals
    // 自定义路由生成
    config.plugins.push(
      new TrRouterPlugin({
        prefix: '@',
        directory: 'src/views/modules',
        routeFilePath: 'src/router/.auto.js'
      })
    )
    // 配合路由meta配置生成 自定义loader
    config.module.rules.push({
      resourceQuery: /blockType=config/,
      loader: require.resolve('@admin/tr-plugin-router/loaders/vue-config-loader.js')
    })
  },

  chainWebpack: config => {
    config.plugins.delete('preload')
    config.plugins.delete('prefetch')

    // 别名 alias
    config.resolve.alias.set('@', resolve('src'))

    // 设置保留空格
    config.module
      .rule('vue')
      .use('vue-loader')
      .loader('vue-loader')
      .tap(options => {
        options.compilerOptions.preserveWhitespace = true
        return options
      })
      .end()
    // 打包分析
    if (IS_PROD) {
      config.plugin('webpack-report').use(BundleAnalyzerPlugin, [
        {
          analyzerMode: 'static'
        }
      ])
    }
    config
      // https://webpack.js.org/configuration/devtool/#development
      .when(!IS_PROD, config => config.devtool('cheap-source-map'))

    config.when(IS_PROD, config => {
      config.optimization.splitChunks({
        chunks: 'all',
        cacheGroups: {
          // cacheGroups 下可以可以配置多个组，每个组根据test设置条件，符合test条件的模块
          commons: {
            name: 'chunk-commons',
            test: resolve('src/components'),
            minChunks: 3, //  被至少用三次以上打包分离
            priority: 5, // 优先级
            reuseExistingChunk: true // 表示是否使用已有的 chunk，如果为 true 则表示如果当前的 chunk 包含的模块已经被抽取出去了，那么将不会重新生成新的。
          },
          node_vendors: {
            name: 'chunk-libs',
            chunks: 'initial', // 只打包初始时依赖的第三方
            test: /[\\/]node_modules[\\/]/,
            priority: 10
          },
          vantUI: {
            name: 'chunk-vantUI', // 单独将 vantUI 拆包
            priority: 20, // 数字大权重到，满足多个 cacheGroups 的条件时候分到权重高的
            test: /[\\/]node_modules[\\/]_?vant(.*)/
          }
        }
      })
      config.optimization.runtimeChunk('single')
    })
  }
}

这个 Webpack 配置是针对 Vue 项目的定制化打包配置，结合了性能优化、开发体验、生产环境适配和特定业务需求（如路由自动生成）。以下是对其关键部分的详细分析，包括优化点、潜在问题和改进建议：

一、核心配置分析 ​

1. 基础配置 ​

  • publicPath: '/baseplatformh5'

    • 适用于项目部署在非根路径（如 https://example.com/baseplatformh5），需确保与路由的 base 配置一致。

    • 问题

      ：如果部署路径动态变化（如多环境不同路径），建议通过环境变量动态设置：

      javascript
      javascript

publicPath: process.env.VUE_APP_PUBLIC_PATH || '/'
      1
      2
      3

  • externals 排除 CDN 依赖

    • 将 vue、vue-router、vant 等库通过 CDN 引入，减少打包体积。
    • 风险：需确保 CDN 资源稳定，否则需提供回退方案（如本地备份）。

  • pages 多页面配置

    • 当前仅配置单页入口（index），但支持扩展多页应用（MPA）。
    • 改进：动态生成多页配置（如根据 src/pages 目录自动注册）。

2. 开发环境优化 ​

  • devServer 配置

    • 代理 /gateway 到测试服务器，解决跨域问题。

    • 建议：通过环境变量切换代理目标，避免硬编码：

      javascript
      javascript

target: process.env.VUE_APP_API_PROXY || 'http://default-api.com'
      1
      2
      3

    • headers: { 'Access-Control-Allow-Origin': '\*' } 允许跨域请求，但生产环境应移除（由后端配置 CORS）。

  • cheap-source-map

    • 开发环境使用快速源码映射，平衡构建速度和调试体验。

3. 生产环境优化 ​

(1) 代码分割 (splitChunks) ​

  • 策略：
    • chunk-commons: 提取 src/components 中被至少 3 个入口引用的模块。
    • chunk-libs: 分离 node_modules 中的初始依赖。“chunk-libs: 分离 node_modules 中的初始依赖” 是 Webpack 等打包工具中通过 splitChunks 配置实现的一种优化策略，其核心目的是将项目初始加载时所需的第三方依赖库（来自 node_modules）从主包中拆分出来，形成独立的代码块（chunk），从而提升应用的加载性能和缓存利用率。
    • chunk-vantUI: 单独打包 Vant UI 库（高优先级）。
  • 优点：减少重复代码，利用浏览器缓存。
  • 潜在问题：
    • minChunks: 3 可能过于严格，导致部分公共代码未被提取。
    • 建议：根据项目规模调整阈值（如 minChunks: 2）。

(2) 运行时优化 (runtimeChunk) ​

  • runtimeChunk: 'single'
    • 将 Webpack 运行时逻辑提取到单独文件，避免因入口文件变化导致缓存失效。

(3) CSS 提取 ​

  • extract: IS_PROD
    • 生产环境提取 CSS 为独立文件，利用缓存。
    • 问题：若项目样式较少，动态注入（extract: false）可能更高效。

4. 特色功能 ​

(1) 路由自动生成 (TrRouterPlugin) ​

  • 作用：扫描 src/views/modules 目录，自动生成路由配置到 src/router/.auto.js。
  • 优势：减少手动维护路由的工作量。
  • 风险：
    • 需确保目录结构符合插件预期（如文件命名规则）。
    • 动态路由可能无法处理复杂权限逻辑。

(2) 自定义 Loader (vue-config-loader) ​

  • 用途：通过 resourceQuery: /blockType=config/ 解析特殊模块（如路由元信息）。
  • 示例：可能在 Vue 文件中通过 <script blockType="config"> 定义路由配置。

(3) 打包分析 (BundleAnalyzerPlugin) ​

  • 生产环境静态报告：通过 analyzerMode: 'static' 生成分析文件（dist/report.html）。
  • 建议：开发环境也可启用交互式分析（analyzerMode: 'server'）。

1. 性能优化 ​

  • Gzip 压缩缺失 建议添加 compression-webpack-plugin 压缩静态资源（需服务器配合解压）。

  • Preload/Prefetch 移除

configureWebpack ​

chainWebpack 和 configureWebpack 的区别

在 Vue CLI 项目中，chainWebpack 和 configureWebpack 都是用于自定义 Webpack 配置的选项，但它们的实现方式和适用场景有所不同。以下是两者的详细对比：

1. configureWebpack ​

特点 ​

  • 直接修改配置：通过对象或函数直接合并或覆盖 Webpack 配置。
  • 简单易用：适合快速添加或修改配置项（如插件、别名、环境变量等）。
  • 合并规则：
    • 对象形式：与 Vue CLI 默认配置深度合并（相同属性会覆盖，新增属性会追加）。
    • 函数形式：接收默认配置作为参数，可动态修改后返回新配置。

适用场景 ​

  • 添加全局变量（DefinePlugin）。
  • 配置路径别名（resolve.alias）。
  • 拆分第三方库（splitChunks）。
  • 排除特定模块（IgnorePlugin）。

示例 ​

(1) 对象形式 ​

javascript
// vue.config.js
module.exports = {
  configureWebpack: {
    resolve: {
      alias: {
        '@utils': '@/utils' // 添加路径别名
      }
    },
    plugins: [
      new webpack.DefinePlugin({
        __APP_VERSION__: JSON.stringify('1.0.0') // 定义全局变量
      })
    ]
  }
}

2. chainWebpack ​

特点 ​

  • 链式操作：基于 webpack-chain 提供链式 API，可精细控制配置。
  • 灵活性强：适合修改 Loader 规则、插件参数、条件判断等复杂场景。
  • 不可变性：所有修改通过链式调用完成，避免直接操作对象导致的副作用。

适用场景 ​

  • 修改 Babel/TypeScript Loader 配置。
  • 添加新的文件类型 Loader（如 .md、.svg）。
  • 根据环境动态调整规则（如开发环境禁用压缩）。
  • 修改插件的细粒度参数（如 HtmlWebpackPlugin 的模板文件）。

示例 ​

(1) 修改 Babel Loader 配置 ​

javascript
module.exports = {
  chainWebpack: config => {
    config.module
      .rule('js')
      .test(/\.js$/)
      .exclude.add(/node_modules/) // 排除 node_modules
      .end()
      .use('babel-loader')
      .loader('babel-loader')
      .tap(options => ({
        ...options,
        plugins: ['@babel/plugin-proposal-optional-chaining'] // 添加 Babel 插件
      }))
  }
}

3. 核心区别 ​

特性configureWebpackchainWebpack
操作方式直接修改对象或返回新对象链式调用（基于 webpack-chain）
灵活性适合简单配置适合复杂、精细控制
合并规则深度合并（可能覆盖默认配置）精确修改（不会意外覆盖其他配置）
适用场景添加插件、别名、全局变量修改 Loader、插件参数、条件规则
调试难度较低（直接看对象结构）较高（需理解链式调用）

4. 如何选择？ ​

  • 用 configureWebpack 当：
    • 需要快速添加一个插件或配置项。
    • 配置项是独立的（如别名、全局变量）。
    • 不需要修改 Loader 或插件的内部逻辑。
  • 用 chainWebpack 当：
    • 需要修改 Loader 的选项（如 Babel、Sass）。
    • 需要根据环境动态调整规则。
    • 需要添加新的文件类型处理逻辑。
    • 需要精确控制插件参数（如 HtmlWebpackPlugin 的模板变量）。

5. 结合使用 ​

两者可以同时使用，configureWebpack 处理简单配置，chainWebpack 处理复杂逻辑：

javascript
module.exports = {
  configureWebpack: {
    resolve: {
      alias: {
        '@': path.resolve(__dirname, 'src')
      }
    }
  },
  chainWebpack: config => {
    config.module
      .rule('scss')
      .test(/\.scss$/)
      .use('sass-loader')
      .loader('sass-loader')
      .tap(options => ({
        ...options,
        additionalData: `$primary-color: #42b983;` // 全局 SCSS 变量
      }))
  }
}

总结 ​

  • configureWebpack：简单直接，适合“配置式”修改。
  • chainWebpack：灵活强大，适合“编程式”修改。
  • 最佳实践：优先用 configureWebpack，复杂场景再用 chainWebpack。

          minChunks: 3, //  被至少用三次以上打包分离
          priority: 5, // 优先级
          reuseExistingChunk: true // 表示是否使用已有的 chunk，如果为 true 则表示如果当前的 chunk 包含的模块已经被抽取出去了，那么将不会重新生成新的。
        },
        node_vendors: {
          name: 'chunk-libs',
          chunks: 'initial', // 只打包初始时依赖的第三方
          test: /[\\/]node_modules[\\/]/,
          priority: 10
        },
        vantUI: {
          name: 'chunk-vantUI', // 单独将 vantUI 拆包
          priority: 20, // 数字大权重到，满足多个 cacheGroups 的条件时候分到权重高的
          test: /[\\/]node_modules[\\/]_?vant(.*)/
        }
      }
    })
    config.optimization.runtimeChunk('single')
  })
}

完整优化示例 ​

javascript
const { BundleAnalyzerPlugin } = require('webpack-bundle-analyzer')
const CompressionPlugin = require('compression-webpack-plugin')
const path = require('path')

function resolve(dir) {
  return path.join(__dirname, dir)
}

module.exports = {
  chainWebpack: config => {
    // 1. 移除默认插件
    config.plugins.delete('preload').delete('prefetch')

    // 2. 别名配置
    config.resolve.alias.set('@', resolve('src')).set('@api', resolve('src/api'))

    // 3. Vue 模板空格保留（Vue 2）
    config.module
      .rule('vue')
      .use('vue-loader')
      .tap(options => {
        options.compilerOptions.preserveWhitespace = true
        return options
      })

    // 4. 打包分析
    if (process.env.ANALYZE) {
      config.plugin('webpack-report').use(BundleAnalyzerPlugin, [{ analyzerMode: 'static' }])
    }

    // 5. Source Map
    config.when(!IS_PROD, config => config.devtool('cheap-source-map'))

    // 6. 生产环境优化
    config.when(IS_PROD, config => {
      // 代码分割
      config.optimization.splitChunks({
        chunks: 'all',
        cacheGroups: {
          libs: {
            name: 'chunk-libs',
            test: /[\\/]node_modules[\\/]/,
            priority: 10
          },
          vant: {
            name: 'chunk-vant',
            test: /[\\/]node_modules[\\/]_?vant(.*)/,
            priority: 20
          },
          commons: {
            name: 'chunk-commons',
            test: resolve('src/components'),
            minChunks: 3,
            priority: 5,
            reuseExistingChunk: true
          }
        }
      })

      // 提取 runtime 代码
      config.optimization.runtimeChunk('single')

      // Gzip 压缩
      config.plugin('gzip').use(CompressionPlugin, [
        {
          test: /\.(js|css)$/,
          threshold: 10240
        }
      ])
    })
  }
}

总结 ​
